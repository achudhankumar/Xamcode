if(!reduce){
  document.body.classList.add('hero-ready');

var W=800,H=600,mx=.5,my=.4,t=0;var pts=[];

cv.width=W;cv.height=H;pts=[];var n=Math.floor(Math.min(90,Math.max(36,W/16)));for(var i=0;i<n;i++){pts.push({x:Math.random()*W,y:Math.random()*H,vx:(Math.random()-.5)*.35,vy:(Math.random()-.5)*.35,r:Math.random()*1.8+.6});}}

function frame(){t+=.008;try{ctx.clearRect(0,0,W,H);}catch(e){return;}
try{
var gx=W*(.3+.4*mx),gy=H*(.35+.3*my);
var g=ctx.createRadialGradient(gx,gy,0,gx,gy,Math.max(W,H)*.7);g.addColorStop(0,'rgba(200,255,61,.10)');g.addColorStop(.4,'rgba(91,108,255,.08)');g.addColorStop(1,'rgba(11,14,7,0)');ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
ctx.strokeStyle='rgba(200,255,61,.16)';ctx.lineWidth=1;
for(var i=0;i<pts.length;i++){var a=pts[i];
if(!reduce){a.x+=a.vx+(mx-.5)*.4;a.y+=a.vy+(my-.4)*.4;if(a.x<0)a.x=W;if(a.x>W)a.x=0;if(a.y<0)a.y=H;if(a.y>H)a.y=0;}
for(var j=i+1;j<pts.length;j++){var b=pts[j];var dx=a.x-b.x,dy=a.y-b.y;var d=Math.sqrt(dx*dx+dy*dy);if(d<130){ctx.globalAlpha=Math.max(0,(130-d)/130*.5);ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.stroke();}}
ctx.globalAlpha=.9;ctx.fillStyle=i%5===0?'#C8FF3D':'rgba(241,239,227,.7)';ctx.beginPath();ctx.arc(a.x+Math.sin(t+i)*4,a.y+Math.cos(t*.8+i)*4,a.r,0,7);ctx.fill();}
ctx.globalAlpha=1;}catch(e){}
if(!reduce){try{if(typeof requestAnimationFrame!=='undefined')requestAnimationFrame(frame);}catch(e){}}}
try{frame();}catch(e){}
}catch(e){}}

try{if('IntersectionObserver' in window){var io=new IntersectionObserver(function(es){es.forEach(function(en){if(en.isIntersecting){var id=en.target.id;for(var k=0;k<links.length;k++){var on=links[k].getAttribute('href')==='#'+id;if(on)links[k].classList.add('act');else links[k].classList.remove('act');}}});},{rootMargin:'-40% 0px -55% 0px'});

function cineScroll(){try{
var track=$('cineTrack'),cam=$('cineCamera'),hud=$('cineHud');if(!track)return;
var acts=[$('act1'),$('act2'),$('act3'),$('act4'),$('act5'),$('act6')].filter(Boolean);
var actLabel=$('chActLabel'),actBar=$('chProgFill'),coords=$('chCoords');
var actNames=['ACT 01 · CURRENT STATE','ACT 02 · STUDY ENGINE','ACT 03 · AURA AI TUTOR','ACT 04 · ROADMAP','ACT 05 · LAUNCH READINESS','ACT 06 · FINAL REVEAL'];

var cineRevs=document.querySelectorAll('.cine-reveal');
var reduce=false;
try{reduce=window.matchMedia&&window.matchMedia('(prefers-reduced-motion: reduce)').matches;}catch(e){}

if(!reduce){
  document.body.classList.add('js-cine');
}

if('IntersectionObserver' in window && !reduce){
  var revIo=new IntersectionObserver(function(entries){
    entries.forEach(function(en){
      if(en.isIntersecting){
        en.target.classList.add('in-view');
        revIo.unobserve(en.target);
      }
    });
  },{threshold:0.12, rootMargin:'0px 0px -40px 0px'});
  cineRevs.forEach(function(r){revIo.observe(r);});

  var actIo=new IntersectionObserver(function(entries){
    entries.forEach(function(en){
      if(en.isIntersecting){
        en.target.classList.add('act-entered');
      }
    });
  },{threshold:0.05});
  acts.forEach(function(a){actIo.observe(a);});
}else{
  cineRevs.forEach(function(r){r.classList.add('in-view');});
  acts.forEach(function(a){a.classList.add('act-entered');});
}

function onScroll(){try{
  var winH=window.innerHeight||800;
  var trackRect=track.getBoundingClientRect();
  var trackH=track.offsetHeight||1;
  var inTrack=trackRect.top<winH*0.75&&trackRect.bottom>winH*0.2;
  if(hud){
    if(inTrack&&window.innerWidth>900){
      hud.classList.add('hud-visible');
    }else{
      hud.classList.remove('hud-visible');
    }
  }

  var progress=Math.max(0,Math.min(1,(-trackRect.top+winH*0.25)/trackH));
  if(actBar)actBar.style.width=(progress*100).toFixed(1)+'%';

  var curActIdx=0;
  for(var i=0;i<acts.length;i++){
    var r=acts[i].getBoundingClientRect();
    if(r.top<=winH*0.55&&r.bottom>=winH*0.25){
      curActIdx=i;
    }
  }
  for(var k=0;k<acts.length;k++){
    if(k===curActIdx){
      acts[k].classList.add('act-active');
    }else{
      acts[k].classList.remove('act-active');
    }
  }
  if(actLabel&&actNames[curActIdx])actLabel.textContent=actNames[curActIdx];
  if(coords)coords.textContent='ACT: '+(curActIdx+1)+'/6 · '+(progress*100).toFixed(0)+'% SCROLL';

  if(!reduce && window.innerWidth > 900){
    var y = window.scrollY || 0;
    var heroT = $('heroTitle');
    var heroCv = $('heroCanvas');
    var bgGlow = document.querySelector('.bg.glow') || document.querySelector('.bg-glow');
    if(heroT && y < 1400){
      heroT.style.transform = 'translate3d(0, -' + Math.min(50, (y * 0.09)).toFixed(1) + 'px, 0)';
    }
    if(heroCv && y < 1400){
      heroCv.style.transform = 'translate3d(0, ' + Math.min(60, (y * 0.12)).toFixed(1) + 'px, 0)';
    }
    if(bgGlow){
      bgGlow.style.transform = 'translate3d(0, -' + (y * 0.025).toFixed(1) + 'px, 0)';
    }
    var act1Gauge = document.querySelector('#act1 .cw');
    if(act1Gauge && trackRect.top < winH && trackRect.bottom > 0){
      var gOffset = Math.max(-18, Math.min(18, (trackRect.top / winH) * 14));
      act1Gauge.style.transform = 'translate3d(0, ' + gOffset.toFixed(1) + 'px, 0)';
    }
  }
}catch(x){}}

window.addEventListener('scroll',onScroll,{passive:true});
window.addEventListener('resize',onScroll,{passive:true});
onScroll();

      window.scrollTo({top:Math.max(0,targetTop),behavior:'smooth'});
    }
  }catch(err){}});
}
}catch(e){}}

})();
