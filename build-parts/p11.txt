function rReadiness(s){
  s=s||{};
  function set(n,v){var b=$(n);if(!b)return v;b.textContent=v;return v;}
  var ph=(s.progress&&s.progress.phase)?s.progress.phase:'—';
  set('lrStage',ph);
  set('lrStagePill','verified weight');
  if(!ph)set('lrStage','—');
  var latest=null;
  (s.milestones||[]).forEach(function(m){
    if(!latest&&(m.status==='verified'||m.status==='done'||m.status==='completed'))latest=m;
  });
  if(latest){
    set('lrVerified',latest.title||'—');
    set('lrVerifiedPill','verified');
    set('lrVerifiedMeta','Verified and recorded '+fm(latest.verifiedAt||''));
  }else{
    set('lrVerified','—');
    set('lrVerifiedPill','none');
    set('lrVerifiedMeta','No verified milestone recorded yet.');
  }
  var nx=null;
  (s.milestones||[]).forEach(function(m){
    if(!nx&&(m.status==='in_progress'||m.status==='in-progress'||m.status==='active'))nx=m;
  });
  if(!nx){
    (s.milestones||[]).forEach(function(m){
      if(!nx&&(m.status==='planned'||m.status==='upcoming'||m.status==='todo'))nx=m;
    });
  }
  if(nx){
    set('lrNext',nx.title||'—');
    set('lrNextPill','in_progress');
    set('lrNextMeta','Unverified until completion criteria pass.');
  }else{
    set('lrNext','—');
    set('lrNextPill','none');
    set('lrNextMeta','No next development milestone yet.');
  }
  var lap='pre-launch';
  var lpad='';
  var l=s.launch||{};
  if(l.hasConfirmedDate&&s.launch.readinessPercent>=100){
    lap='ready to launch';
  }else if(s.launch.readinessPercent!=null){
    lap='building toward launch';
    lpad=Math.round(s.launch.readinessPercent)+'% verified';
  }else if(l.hasConfirmedDate){
    lap='pending confirmation';
  }
  set('lrLaunch',lap);
  set('lrLaunchPill',lap==='ready to launch'?'ready':(lap==='building toward launch'?'in_progress':'pre-launch'));
  set('lrLaunchMeta',lpad||'All release milestones must be verified before launch.');
  set('lrUpdateTitle','Verified Milestone & Development Log');
  set('lrUpdateBadge','public record');
}

function loadIdeaComments(ideaId,btn,box){
  if(!box)return;
  box.textContent='';
  box.appendChild(el('p','sub','Loading comments…'));
  gt('/api/bip/ideas/'+encodeURIComponent(ideaId)+'/comments').then(function(r){
    var list=(r&&r.comments)||[];
    if(btn)btn.textContent='💬 '+list.length+' comments';
    box.textContent='';
    var ul=el('ul','idea-cmt-list');
    if(!list.length){
      var empty=el('li','sub','No comments yet — add the first reply below.');
      ul.appendChild(empty);
    }else{
      list.forEach(function(c){
        var li=el('li','idea-cmt-item');
        var hd=el('div','idea-cmt-head');
        hd.appendChild(el('strong',null,c.nickname||'Visitor'));
        var right=el('div',null);
        right.appendChild(el('span',null,fm(c.at)+' '));
        var rbtn=el('button','vote-btn',null);
        rbtn.type='button';
        rbtn.style.padding='2px 8px';
        rbtn.style.fontSize='10px';
        rbtn.textContent='Report';
        (function(cid){rbtn.addEventListener('click',function(){report('comment',cid);});})(c.id);
        right.appendChild(rbtn);
        hd.appendChild(right);
        li.appendChild(hd);
        li.appendChild(el('div',null,c.text||''));
        ul.appendChild(li);
      });
    }
    box.appendChild(ul);
    var fmEl=el('form','idea-cmt-form');
    var nickIn=el('input',null);nickIn.type='text';nickIn.maxLength=40;nickIn.placeholder='Name (optional)';
    var txtIn=el('input',null);txtIn.type='text';txtIn.maxLength=1000;txtIn.required=true;txtIn.placeholder='Write a comment…';
    var subBtn=el('button','btn primary','Reply');subBtn.type='submit';subBtn.style.minHeight='36px';subBtn.style.padding='0 14px';subBtn.style.marginTop='0';subBtn.style.fontSize='12px';
    fmEl.appendChild(nickIn);fmEl.appendChild(txtIn);fmEl.appendChild(subBtn);
    fmEl.addEventListener('submit',function(ev){
      ev.preventDefault();
      if(!txtIn.value||!txtIn.value.trim())return;
      subBtn.disabled=true;
      pt('/api/bip/ideas/'+encodeURIComponent(ideaId)+'/comments',{nickname:nickIn.value,text:txtIn.value}).then(function(){
        txtIn.value='';

      }).catch(function(e){
        window.alert((e&&e.message)||'Could not post comment');
      }).then(function(){subBtn.disabled=false;});
    });
    box.appendChild(fmEl);
  }).catch(function(e){
    box.textContent='';
    box.appendChild(el('p','sub',(e&&e.message)||'Could not load comments.'));
  });
}
